% Conventus document skeleton — ONE trusted, in-repo LaTeX skeleton shared by
% all three document kinds (invoice / receipt / adjustment note). Option A:
% admins never author raw LaTeX; the renderer substitutes escaped structured
% content into the tokens below. Never place admin-authored content here.
%
% SUBSTITUTION CONTRACT — the renderer (app/services/documents.py::assemble_tex)
% replaces every token of the form @@name@@ before compiling. The @@...@@
% delimiter cannot occur in normal LaTeX, so ordinary braces here stay
% unambiguous. (The token NAMES are listed bare below — never as literal
% tokens — so a multi-line substituted value can't leak out of this comment.)
%
%   Boolean flags — raw \...true / \...false control sequences:
%     flag_gst, flag_gst_registered, flag_logo, flag_signature, flag_recipient_abn,
%     flag_recipient_address, flag_due_date, flag_business_number,
%     flag_business_address, flag_contact_email, flag_signatory,
%     flag_payment_instructions, flag_payment_link, flag_pdf_body,
%     flag_invoice, flag_receipt, flag_adjustment
%   Asset filenames — raw basenames of files copied into the job dir:
%     logo_file, signature_file
%   Text values — LaTeX-escaped (unless RawLatex):
%     doc_title, site_name, business_legal_name, business_number,
%     business_address, business_contact_email, signatory_name,
%     signatory_role, user_name, user_email, recipient_abn, recipient_address,
%     conference_title, conference_dates, tier_name, transaction_id,
%     registration_id, payment_date, due_date, currency_symbol, currency_code,
%     amount, gst_amount, amount_ex_gst, gst_rate, payment_instructions,
%     payment_link, pdf_body
%
% Packages are limited to those verified warm in the tectonic cache:
% geometry, fontenc, helvet, graphicx, array, tabularx, xcolor.

% --- Header: title + number/date (left), logo + issuer details (right) ----
\noindent
% The split favours the issuer block: the left column's longest line is a
% fixed-width label plus a reference ("Adjustment Number: INV-20260806-28D4",
% ~68mm at 11pt), so it has room to spare at 0.46, while the address on the
% right is free text a society can make as long as its postal address needs.
\begin{minipage}[t]{0.46\linewidth}
  \vspace{0pt}
  % Ragged, not justified: the longest label/reference pair ("Adjustment
  % Number:" plus a widened reference) does not fit on one line, and
  % justification stretched the interword spaces across the whole column to
  % hide it. \mbox keeps the reference itself in one piece — it broke at its
  % own hyphen otherwise, which is unreadable on a document whose entire
  % purpose is to be quoted back.
  \raggedright
  {\LARGE\bfseries\MakeUppercase{@@doc_title@@}}\\[6pt]
  {\bfseries\numberlabel:} \mbox{@@transaction_id@@}\\
  {\bfseries\datelabel:} @@payment_date@@%
  \ifisinvoice\ifduedate\\{\bfseries Due Date:} @@due_date@@\fi\fi
\end{minipage}\hfill
\begin{minipage}[t]{0.50\linewidth}
  \vspace{0pt}
  \raggedleft

% --- Tax statement -------------------------------------------------------
% When no GST was charged, say so plainly. A silent omission leaves the payer
% guessing, and a zero-valued GST line would be worse — it implies a taxable
% sale on which no tax was charged.
%
% WHY the two branches: "no GST on this sale" and "this issuer is not
% registered for GST" are different claims. A registered issuer may charge no
% GST on a particular supply (an overseas sponsor, say) and must NOT print a
% non-registration statement on that document — it would be false.
\noindent
\ifgst\else
  \ifgstreg
    {\color{muted}No GST has been charged on this supply.}
  \else
    {\color{muted}No GST has been charged. @@business_legal_name@@ is not
     registered for GST.}
  \fi
\fi

% --- Sign-off ------------------------------------------------------------
% Wrapped in a minipage because a minipage cannot break across pages. Left
% loose, the block split wherever the page ran out — most often stranding the

% the first. A signature separated from the entity it signs for is worse than
% a page break, so the whole block now moves or stays as one.
\ifsignatory

  @@business_legal_name@@
