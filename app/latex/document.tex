% Conventus document skeleton — ONE trusted, in-repo LaTeX skeleton shared by
% all three document kinds (invoice / receipt / adjustment note). Option A:
% admins never author raw LaTeX; the renderer substitutes escaped structured
% content into the tokens below. Never place admin-authored content here.
%
% SUBSTITUTION CONTRACT — the renderer (app/services/documents.py::assemble_tex)
% replaces every token of the form @@name@@ before compiling. The @@...@@
% delimiter cannot occur in normal LaTeX, so ordinary braces here stay
% unambiguous. (The token NAMES are listed bare below — never as literal
% tokens — so a multi-line substituted value can't leak out of this comment.)
%
%   Boolean flags — raw \...true / \...false control sequences:
%     flag_gst, flag_logo, flag_signature, flag_recipient_abn,
%     flag_recipient_address, flag_due_date, flag_business_number,
%     flag_business_address, flag_contact_email, flag_signatory,
%     flag_payment_instructions, flag_payment_link, flag_pdf_body,
%     flag_invoice, flag_receipt, flag_adjustment
%   Asset filenames — raw basenames of files copied into the job dir:
%     logo_file, signature_file
%   Text values — LaTeX-escaped (unless RawLatex):
%     doc_title, site_name, business_legal_name, business_number,
%     business_address, business_contact_email, signatory_name,
%     signatory_role, user_name, user_email, recipient_abn, recipient_address,
%     conference_title, conference_dates, tier_name, transaction_id,
%     registration_id, payment_date, due_date, currency_symbol, currency_code,
%     amount, gst_amount, amount_ex_gst, payment_instructions, payment_link,
%     pdf_body
%
% Packages are limited to those verified warm in the tectonic cache:
% geometry, fontenc, helvet, graphicx, array, tabularx, xcolor.

\usepackage[T1]{fontenc}
